import { LABEL } from "./constants";

/**
 * Converts the rendered canvas into the 1-bit monochrome bitmap format
 * TSPL's BITMAP command expects: rows padded to byte boundaries, MSB-first,
 * 0 = black, 1 = white.
 */
export function canvasToMonoBitmap(imageData: ImageData): Uint8Array {
  const { data } = imageData;
  const { width, height } = imageData;
  const widthBytes = Math.ceil(width / 8);
  const bitmap = new Uint8Array(widthBytes * height).fill(255);

  for (let py = 0; py < height; py++) {
    for (let px = 0; px < width; px++) {
      const idx = (py * width + px) * 4;
      if (data[idx] < 128 && data[idx + 3] > 128) {
        bitmap[py * widthBytes + (px >> 3)] &= ~(1 << (7 - (px % 8)));
      }
    }
  }
  return bitmap;
}

function concat(chunks: Uint8Array[]): Uint8Array {
  const total = chunks.reduce((n, c) => n + c.length, 0);
  const out = new Uint8Array(total);

import { useCallback, useState } from "react";

/** Minimal shape of the USB device used by the printer flow. */
export type UsbPrinter = USBDevice;

interface UseWebUsbResult {
  device: USBDevice | null;
  isPrinting: boolean;
  error: string | null;
  connect: () => Promise<void>;
  print: (data: BufferSource) => Promise<void>;
}

/**
 * Connects to a label printer over WebUSB and streams raw TSPL bytes to it.
 */
export function useWebUsb(): UseWebUsbResult {
  const [device, setDevice] = useState<USBDevice | null>(null);
  const [isPrinting, setIsPrinting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const connect = useCallback(async () => {
    try {
      if (!navigator.usb) {
        throw new Error("WebUSB is not supported in this browser.");
      }
      const selectedDevice = await navigator.usb.requestDevice({ filters: [] });
      await selectedDevice.open();
      if (selectedDevice.configuration === null) {
        await selectedDevice.selectConfiguration(1);
      }
      await selectedDevice.claimInterface(0);
      setDevice(selectedDevice);
      setError(null);
    } catch (err) {
      console.error("USB connection error:", err);
      setError(err instanceof Error ? err.message : "Failed to connect to printer");
    }
  }, []);

  const print = useCallback(async (data: BufferSource) => {
    if (!device) { setError("No printer connected"); return; }
    setIsPrinting(true);
    setError(null);
    try {
      const endpoint = device.configuration?.interfaces[0].alternate.endpoints.find((ep) => ep.direction === "out");
      if (!endpoint) throw new Error("Could not find a valid USB output endpoint.");
      await device.transferOut(endpoint.endpointNumber, data);
    } catch (err) {
      console.error("Printing error:", err);
      setError(err instanceof Error ? err.message : "Failed to print");
    } finally {
      setIsPrinting(false);
    }
  }, [device]);

  return { device, isPrinting, error, connect, print };
}

import { useCallback, useMemo, useState } from "react";
import type { QueueItem, StoreItem } from "@/lib/catalog";

/** Manages the print queue: add catalog or custom items, adjust quantities, remove. */
export function useLabelQueue() {
  const [queue, setQueue] = useState<QueueItem[]>([]);

  const add = useCallback((item: StoreItem) => {
    setQueue((prev) => {
      const existing = prev.find((q) => q.id === item.id);
      if (existing) {
        return prev.map((q) => (q.id === item.id ? { ...q, quantity: q.quantity + 1 } : q));
      }
      return [...prev, { ...item, quantity: 1 }];
    });
  }, []);

  const setQuantity = useCallback((id: string, quantity: number) => {
    setQueue((prev) => prev.map((q) => (q.id === id ? { ...q, quantity: Math.max(1, quantity) } : q)));
  }, []);

  const adjustQuantity = useCallback((id: string, delta: number) => {
    setQueue((prev) => prev.map((q) => (q.id === id ? { ...q, quantity: Math.max(1, q.quantity + delta) } : q)));
  }, []);

  const remove = useCallback((id: string) => {
    setQueue((prev) => prev.filter((q) => q.id !== id));
  }, []);

  const clear = useCallback(() => setQueue([]), []);

  const totalLabels = useMemo(() => queue.reduce((sum, item) => sum + item.quantity, 0), [queue]);

  return { queue, add, setQuantity, adjustQuantity, remove, clear, totalLabels };
}
